 function valid(item,selectedCountry=country,selectedTab=active){
  const config=countries[selectedCountry];

 function getFeed(selectedCountry,selectedTab,force=false){
  const key=selectedCountry+':'+selectedTab,cached=feeds.get(key);
  if(!force&&cached&&Date.now()-cached.at<30000)return Promise.resolve(cached.data);
  if(pendingFeeds.has(key))return pendingFeeds.get(key).promise;
  const controller=new AbortController(),entry={controller};
  const timeout=setTimeout(()=>controller.abort(),25000);
  entry.promise=(async()=>{
   try{
    const response=await fetch(`${endpoint}/${kinds[selectedTab]}?country=${selectedCountry}`,{signal:controller.signal,cache:'no-cache',credentials:'omit'});
    if(!response.ok)throw new Error('feed');const data=await response.json();
    if(controller.signal.aborted||data.country!==selectedCountry||data.kind!==kinds[selectedTab]||data.truncated||!Array.isArray(data.items)||!data.items.every(item=>valid(item,selectedCountry,selectedTab)))throw new Error('invalid_feed');
    feeds.set(key,{at:Date.now(),data});return data;
   }catch(error){if(pendingFeeds.get(key)===entry)feeds.delete(key);throw error;}
   finally{clearTimeout(timeout);if(pendingFeeds.get(key)===entry)pendingFeeds.delete(key);}
  })();pendingFeeds.set(key,entry);return entry.promise;
 }
