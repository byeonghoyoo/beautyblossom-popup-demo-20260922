 function valid(item,selectedCountry=country,selectedTab=active){
  const config=countries[selectedCountry];
  if(!item||typeof item.title!=='string'||typeof item.image!=='string'||typeof item.href!=='string')return false;
  try{
   const href=new URL(item.href),base=config.paths[selectedTab].replace(/\/$/,'');
   if(href.origin!==config.origin||href.username||href.password)return false;
   const p=href.pathname.replace(/\/$/,'');
   if(selectedCountry==='HK'?!p.startsWith(base+'/'):selectedCountry==='TH'&&selectedTab===2?!['/events','/events/detail'].includes(p):p!==base&&!(selectedCountry==='TH'&&selectedTab===0&&p==='/20'))return false;
   if(item.image){const img=new URL(item.image);if(img.protocol!=='https:'||img.username||img.password)return false;}

  }catch{return false;}
 }
 function feedValid(data,selectedCountry,selectedTab){
  return !!data&&data.country===selectedCountry&&data.kind===kinds[selectedTab]&&!data.truncated&&Array.isArray(data.items)&&data.items.every(item=>valid(item,selectedCountry,selectedTab));

 const feedStorageKey=(selectedCountry,selectedTab)=>`bb-popup-feed-v1:${endpoint}:${selectedCountry}:${kinds[selectedTab]}`;
 function recentFeed(selectedCountry,selectedTab){
  const cached=feeds.get(selectedCountry+':'+selectedTab),now=Date.now();
  if(cached&&now>=cached.at&&now-cached.at<30000)return cached.data;
  try{
   const key=feedStorageKey(selectedCountry,selectedTab),raw=sessionStorage.getItem(key);
   if(!raw)return null;
   if(raw.length<=250000){const saved=JSON.parse(raw);if(Number.isFinite(saved?.at)&&now>=saved.at&&now-saved.at<60000&&feedValid(saved.data,selectedCountry,selectedTab))return saved.data;}
   sessionStorage.removeItem(key);
  }catch{}
  return null;
 }
 function rememberFeed(selectedCountry,selectedTab,data){
  const saved={at:Date.now(),data};feeds.set(selectedCountry+':'+selectedTab,saved);
  try{const raw=JSON.stringify(saved);if(raw.length<=250000)sessionStorage.setItem(feedStorageKey(selectedCountry,selectedTab),raw);}catch{}
 }
 function getFeed(selectedCountry,selectedTab,force=false){
  const key=selectedCountry+':'+selectedTab,cached=!force&&recentFeed(selectedCountry,selectedTab);
  if(cached)return Promise.resolve(cached);
  if(pendingFeeds.has(key))return pendingFeeds.get(key).promise;
  const controller=new AbortController(),entry={controller};
  const timeout=setTimeout(()=>controller.abort(),25000);
  entry.promise=(async()=>{
   try{
    const response=await fetch(`${endpoint}/${kinds[selectedTab]}?country=${selectedCountry}`,{signal:controller.signal,cache:force?'no-cache':'default',credentials:'omit'});
    if(!response.ok)throw new Error('feed');const data=await response.json();
    if(controller.signal.aborted||!feedValid(data,selectedCountry,selectedTab))throw new Error('invalid_feed');
    rememberFeed(selectedCountry,selectedTab,data);return data;
   }catch(error){if(pendingFeeds.get(key)===entry)feeds.delete(key);throw error;}
   finally{clearTimeout(timeout);if(pendingFeeds.get(key)===entry)pendingFeeds.delete(key);}
  })();pendingFeeds.set(key,entry);return entry.promise;
 }
